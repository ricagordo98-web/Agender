```csharp
namespace Agender.Shared.Reports;

public static class ReportPeriod
{
    public static (DateOnly From, DateOnly To) ThisWeek(DateOnly today)
    {
        DateOnly monday = today.AddDays(-(((int)today.DayOfWeek + 6) % 7));
        return (monday, monday.AddDays(6));
    }

    public static (DateOnly From, DateOnly To) LastWeek(DateOnly today) => ThisWeek(today.AddDays(-7));

    public static (DateOnly From, DateOnly To) ThisMonth(DateOnly today)
    {
        DateOnly first = new(today.Year, today.Month, 1);
        return (first, first.AddMonths(1).AddDays(-1));
    }

    public static (DateOnly From, DateOnly To) LastMonth(DateOnly today) =>
        ThisMonth(new DateOnly(today.Year, today.Month, 1).AddDays(-1));
}
```

```csharp
using Agender.Shared.Components;
using Agender.Shared.Domain;

namespace Agender.Shared.Reports;

public static class ReportBuilder
{
    public const string NoProjectLabel = "Sem projeto";
    public const string NoClientLabel = "Sem cliente";
    public const string NoTagLabel = "Sem tag";

    public static int Minutes(TimeEntry entry) => entry.EndMinute - entry.StartMinute;

    public static IReadOnlyList<TimeEntry> Filter(IEnumerable<TimeEntry> entries, Guid? projectId, string? tag) =>
    [
        .. entries.Where(entry =>
            (projectId is null || entry.ProjectId == projectId)
            && (string.IsNullOrEmpty(tag) || entry.Tags.Any(t => string.Equals(t.Name, tag, StringComparison.OrdinalIgnoreCase))))
    ];

    public static ReportResult Build(IReadOnlyList<TimeEntry> entries, IReadOnlyList<Project> projects, ReportGroupBy groupBy)
    {
        Dictionary<Guid, Project> byId = projects.ToDictionary(p => p.Id);

        IReadOnlyList<ReportGroup> groups = groupBy switch
        {
            ReportGroupBy.ClientProject => ByClientProject(entries, byId),
            ReportGroupBy.Project => ByProject(entries, byId),
            ReportGroupBy.Tag => ByTag(entries),
            _ => ByDay(entries)
        };

        return new ReportResult(entries.Sum(Minutes), groups);
    }

    private static Project? Resolve(TimeEntry entry, Dictionary<Guid, Project> byId) =>
        entry.ProjectId is Guid id && byId.TryGetValue(id, out Project? project) ? project : null;

    private static IReadOnlyList<ReportGroup> Sorted(IEnumerable<ReportGroup> groups) =>
    [
        .. groups.OrderByDescending(g => g.Minutes).ThenBy(g => g.Label, StringComparer.OrdinalIgnoreCase)
    ];

    private static IReadOnlyList<ReportGroup> ByProject(IEnumerable<TimeEntry> entries, Dictionary<Guid, Project> byId) =>
        Sorted(entries
            .GroupBy(entry => Resolve(entry, byId)?.Id)
            .Select(group => new ReportGroup(
                group.Key is Guid id ? byId[id].Name : NoProjectLabel,
                group.Sum(Minutes),
                [])));

    private static IReadOnlyList<ReportGroup> ByClientProject(IReadOnlyList<TimeEntry> entries, Dictionary<Guid, Project> byId)
    {
        List<ReportGroup> groups = [];

        IEnumerable<TimeEntry> withProject = entries.Where(entry => Resolve(entry, byId) is not null);

        groups.AddRange(withProject
            .GroupBy(entry => ClientLabel(Resolve(entry, byId)!), StringComparer.OrdinalIgnoreCase)
            .Select(client => new ReportGroup(client.Key, client.Sum(Minutes), ByProject(client, byId))));

        List<TimeEntry> withoutProject = [.. entries.Where(entry => Resolve(entry, byId) is null)];

        if (withoutProject.Count > 0)
        {
            groups.Add(new ReportGroup(NoProjectLabel, withoutProject.Sum(Minutes), []));
        }

        return Sorted(groups);
    }

    private static string ClientLabel(Project project) =>
        string.IsNullOrWhiteSpace(project.ClientName) ? NoClientLabel : project.ClientName.Trim();

    private static IReadOnlyList<ReportGroup> ByTag(IReadOnlyList<TimeEntry> entries) =>
        Sorted(entries
            .SelectMany(entry => TagNames(entry).Select(name => (Name: name, Entry: entry)))
            .GroupBy(pair => pair.Name, StringComparer.OrdinalIgnoreCase)
            .Select(group => new ReportGroup(group.Key, group.Sum(pair => Minutes(pair.Entry)), [])));

    private static IEnumerable<string> TagNames(TimeEntry entry) =>
        entry.Tags.Count == 0 ? new[] { NoTagLabel } : entry.Tags.Select(t => t.Name);

    private static IReadOnlyList<ReportGroup> ByDay(IReadOnlyList<TimeEntry> entries) =>
    [
        .. entries
            .GroupBy(entry => entry.Date)
            .OrderBy(group => group.Key)
            .Select(group => new ReportGroup(DateText.Full(group.Key), group.Sum(Minutes), []))
    ];
}
```

```csharp
using System.Globalization;
using System.Text;
using Agender.Shared.Domain;
using ClosedXML.Excel;

namespace Agender.Shared.Reports;

public static class ReportExporter
{
    private static readonly CultureInfo Culture = CultureInfo.GetCultureInfo("pt-PT");

    private static readonly string[] DetailHeaders =
        ["Data", "Início", "Fim", "Minutos", "Horas", "Projeto", "Cliente", "Tags", "Descrição"];

    public static string Hours(int minutes) => (minutes / 60.0).ToString("0.00", Culture);

    public static byte[] ToCsv(IReadOnlyList<TimeEntry> entries, IReadOnlyList<Project> projects)
    {
        Dictionary<Guid, Project> byId = projects.ToDictionary(p => p.Id);
        StringBuilder builder = new();

        builder.Append(string.Join(';', DetailHeaders)).Append("\r\n");

        foreach (TimeEntry entry in Ordered(entries))
        {
            Project? project = Find(entry, byId);
            int minutes = ReportBuilder.Minutes(entry);

            string[] fields =
            [
                entry.Date.ToString("yyyy-MM-dd", CultureInfo.InvariantCulture),
                TimeSlot.FormatMinute(entry.StartMinute),
                TimeSlot.FormatMinute(entry.EndMinute),
                minutes.ToString(CultureInfo.InvariantCulture),
                Hours(minutes),
                project?.Name ?? string.Empty,
                project?.ClientName ?? string.Empty,
                string.Join(", ", entry.Tags.Select(t => t.Name)),
                entry.Description
            ];

            builder.Append(string.Join(';', fields.Select(Escape))).Append("\r\n");
        }

        return [.. Encoding.UTF8.GetPreamble(), .. Encoding.UTF8.GetBytes(builder.ToString())];
    }

    public static byte[] ToXlsx(IReadOnlyList<TimeEntry> entries, IReadOnlyList<Project> projects, ReportResult summary)
    {
        Dictionary<Guid, Project> byId = projects.ToDictionary(p => p.Id);
        using XLWorkbook workbook = new();

        WriteDetail(workbook.AddWorksheet("Detalhe"), entries, byId);
        WriteSummary(workbook.AddWorksheet("Resumo"), summary);

        using MemoryStream stream = new();
        workbook.SaveAs(stream);
        return stream.ToArray();
    }

    private static void WriteDetail(IXLWorksheet sheet, IReadOnlyList<TimeEntry> entries, Dictionary<Guid, Project> byId)
    {
        for (int column = 0; column < DetailHeaders.Length; column++)
        {
            sheet.Cell(1, column + 1).Value = DetailHeaders[column];
        }

        sheet.Row(1).Style.Font.Bold = true;

        int row = 2;

        foreach (TimeEntry entry in Ordered(entries))
        {
            Project? project = Find(entry, byId);
            int minutes = ReportBuilder.Minutes(entry);

            sheet.Cell(row, 1).Value = entry.Date.ToDateTime(TimeOnly.MinValue);
            sheet.Cell(row, 1).Style.DateFormat.Format = "yyyy-mm-dd";
            sheet.Cell(row, 2).Value = TimeSlot.FormatMinute(entry.StartMinute);
            sheet.Cell(row, 3).Value = TimeSlot.FormatMinute(entry.EndMinute);
            sheet.Cell(row, 4).Value = minutes;
            sheet.Cell(row, 5).Value = minutes / 60.0;
            sheet.Cell(row, 5).Style.NumberFormat.Format = "0.00";
            sheet.Cell(row, 6).Value = project?.Name ?? string.Empty;
            sheet.Cell(row, 7).Value = project?.ClientName ?? string.Empty;
            sheet.Cell(row, 8).Value = string.Join(", ", entry.Tags.Select(t => t.Name));
            sheet.Cell(row, 9).Value = entry.Description;
            row++;
        }

        sheet.Columns().AdjustToContents(1, row);
    }

    private static void WriteSummary(IXLWorksheet sheet, ReportResult summary)
    {
        sheet.Cell(1, 1).Value = "Grupo";
        sheet.Cell(1, 2).Value = "Minutos";
        sheet.Cell(1, 3).Value = "Horas";
        sheet.Row(1).Style.Font.Bold = true;

        int row = 2;

        foreach (ReportGroup group in summary.Groups)
        {
            WriteSummaryRow(sheet, row++, group, indent: 0);

            foreach (ReportGroup child in group.Children)
            {
                WriteSummaryRow(sheet, row++, child, indent: 1);
            }
        }

        WriteSummaryRow(sheet, row, new ReportGroup("Total", summary.TotalMinutes, []), indent: 0);
        sheet.Row(row).Style.Font.Bold = true;
        sheet.Columns().AdjustToContents(1, row);
    }

    private static void WriteSummaryRow(IXLWorksheet sheet, int row, ReportGroup group, int indent)
    {
        sheet.Cell(row, 1).Value = group.Label;
        sheet.Cell(row, 1).Style.Alignment.Indent = indent;
        sheet.Cell(row, 2).Value = group.Minutes;
        sheet.Cell(row, 3).Value = group.Minutes / 60.0;
        sheet.Cell(row, 3).Style.NumberFormat.Format = "0.00";
    }

    private static IEnumerable<TimeEntry> Ordered(IEnumerable<TimeEntry> entries) =>
        entries.OrderBy(e => e.Date).ThenBy(e => e.StartMinute);

    private static Project? Find(TimeEntry entry, Dictionary<Guid, Project> byId) =>
        entry.ProjectId is Guid id && byId.TryGetValue(id, out Project? project) ? project : null;

    private static string Escape(string value) =>
        value.IndexOfAny([';', '"', '\r', '\n']) >= 0 ? $"\"{value.Replace("\"", "\"\"")}\"" : value;
}
```

```razor
@code {
    private DateOnly _from;
    private DateOnly _to;
    private ReportGroupBy _groupBy = ReportGroupBy.ClientProject;
    private Guid? _projectId;
    private string? _tag;
    private IReadOnlyList<TimeEntry> _entries = [];
    private IReadOnlyList<TimeEntry> _filtered = [];
    private ReportResult _result = new(0, []);
    private bool _loading = true;
    private IJSObjectReference? _module;

    protected override async Task OnInitializedAsync()
    {
        await Store.InitializeAsync();
        (_from, _to) = ReportPeriod.ThisMonth(Store.Today);
        await LoadAsync();
    }

    private Task SetPeriodAsync((DateOnly From, DateOnly To) period)
    {
        (_from, _to) = period;
        return LoadAsync();
    }

    private async Task LoadAsync()
    {
        if (_to < _from)
        {
            _entries = [];
            Rebuild();
            return;
        }

        _loading = true;

        try
        {
            _entries = await TimeEntryService.GetRangeAsync(_from, _to);
        }
        catch (Exception ex) when (ex is ServerUnavailableException or PersistenceException or UnauthorizedAccessException)
        {
            _entries = [];
            Toasts.Show("Não foi possível carregar os registos.", ToastKind.Error);
        }

        _loading = false;
        Rebuild();
    }

    private void Rebuild()
    {
        _filtered = ReportBuilder.Filter(_entries, _projectId, _tag);
        _result = ReportBuilder.Build(_filtered, Store.Projects, _groupBy);
    }

    private Task ExportCsvAsync() =>
        DownloadAsync("csv", "text/csv", () => ReportExporter.ToCsv(_filtered, Store.Projects));

    private Task ExportXlsxAsync() =>
        DownloadAsync("xlsx", "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
            () => ReportExporter.ToXlsx(_filtered, Store.Projects, _result));

    private async Task DownloadAsync(string extension, string contentType, Func<byte[]> build)
    {
        try
        {
            _module ??= await JS.InvokeAsync<IJSObjectReference>("import", "./_content/Agender.Shared/Pages/Reports.razor.js");
            using MemoryStream stream = new(build());
            using DotNetStreamReference reference = new(stream);
            await _module.InvokeVoidAsync("downloadFile", $"agender-{_from:yyyy-MM-dd}_{_to:yyyy-MM-dd}.{extension}", contentType, reference);
        }
        catch (JSException)
        {
            Toasts.Show("Não foi possível exportar o ficheiro.", ToastKind.Error);
        }
    }

    public async ValueTask DisposeAsync()
    {
        if (_module is not null)
        {
            try
            {
                await _module.DisposeAsync();
            }
            catch (JSDisconnectedException)
            {
            }
        }
    }
}
```
